// MCP client over stdio: starts the installed bin from the consumer directory and lists its tools.
const { createRequire } = require('node:module')
const [, , repo, bin, cwd] = process.argv
const req = createRequire(repo + '/package.json')
const { Client } = req('@modelcontextprotocol/sdk/client/index.js')
const { StdioClientTransport, getDefaultEnvironment } = req('@modelcontextprotocol/sdk/client/stdio.js')
;(async () => {
  const transport = new StdioClientTransport({ command: bin, args: [], cwd, env: getDefaultEnvironment(), stderr: 'pipe' })
  let stderr = ''
  transport.stderr?.on('data', (d) => (stderr += d))
  const client = new Client({ name: 'install-proof', version: '0.0.0' })
  await client.connect(transport)
  const v = client.getServerVersion()
  const { tools } = await client.listTools()
  console.log(`server: ${v.name} ${v.version}`)
  console.log(`tools (${tools.length}): ${tools.map((t) => t.name).join(', ')}`)
  console.log(`stderr: ${stderr.trim().split('\n')[0]}`)
  await client.close()
})().catch((e) => { console.log('FAILED:', e.message); process.exit(1) })
